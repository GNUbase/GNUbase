# ThreadSanitizer suppressions for CI's tsan job (TSAN_OPTIONS=suppressions=...). TSan only sees the locking of code
# compiled with it, so it reports races inside Mesa's lavapipe Vulkan driver, and the LLVM it compiles shaders with,
# which the window tests draw through. Nothing of GNUbase's own code belongs here.
race:libvulkan_lvp.so
race:libLLVM
# The same blindness for libstdc++ itself, which isn't compiled with TSan: an exception's message is a reference-counted
# string whose count it changes inside libstdc++.so, so when a promise's exception is read on one thread and released
# with the promise on another, TSan sees the release as a race with the read (HostManager::load in the safe mode test).
race:std::runtime_error::~runtime_error
